// TRONG FILE admin.html function checkAuthCallback() { const urlParams = new URLSearchParams(window.location.search); // Bắt callback từ OAuth LẦN ĐẦU (auth_status) hoặc callback CHECK STATUS (status_checked) const isCallback = urlParams.get('auth_status') === 'success'; const isStatusRefresh = urlParams.get('status_checked') === 'true'; if (isCallback || isStatusRefresh) { localStorage.setItem('admin_is_logged_in', 'true'); // Luôn cập nhật email (email luôn có mặt trong cả 2 luồng) localStorage.setItem('admin_email', urlParams.get('user_email')); try { // 🔥🔥 ĐIỂM SỬA CHỮA CHÍNH: Cập nhật Tên và Avatar chỉ khi tham số tồn tại // 1. Cập nhật TÊN const userNameFromUrl = urlParams.get('user_name'); // Nếu tham số user_name CÓ TỒN TẠI trong URL (tức là từ luồng OAuth) thì mới ghi đè if (userNameFromUrl !== null) { localStorage.setItem('admin_name', decodeURIComponent(userNameFromUrl)); } // 2. Cập nhật AVATAR const userAvatarFromUrl = urlParams.get('user_avatar'); // Nếu tham số user_avatar CÓ TỒN TẠI trong URL thì mới ghi đè if (userAvatarFromUrl !== null) { localStorage.setItem('admin_avatar', decodeURIComponent(userAvatarFromUrl)); } // 3. Cập nhật Trạng thái và Thời gian HẾT HẠN MỚI NHẤT // Dùng || '0' hoặc || 'new' để đảm bảo luôn có giá trị mặc định nếu n8n không trả về. localStorage.setItem('admin_expiry', urlParams.get('user_expiry') || '0'); localStorage.setItem('admin_status', urlParams.get('user_status') || 'new'); } catch(e) { console.error("Lỗi khi đọc tham số URL:", e); } // Dọn dẹp URL và cập nhật UI history.replaceState(null, null, window.location.pathname); updateUI(); } }
